kc_concord_floodworks_location_eligible = {
	is_land = yes
	region = region:japan_region
	OR = {
		is_coastal = yes
		has_river = yes
		is_adjacent_to_lake = yes
		topography = wetlands
		# The irrigation whitelist represents river plains and enclosed basins
		# even where the base map carries no river or wetland flag.
		kc_concord_irrigation_location_eligible = yes
	}
}

kc_concord_seto_harbor_location_eligible = {
	is_land = yes
	is_coastal = yes
	region = region:japan_region
	OR = {
		# The initial authorization is the historical Inland Sea chain.
		any_neighbor_location = {
			province_definition = province_definition:seto_sea
		}
		# The first late-maritime authorization admits seven established outer
		# ports. Tsuruga is deliberately included as the Sea-of-Japan hinge.
		AND = {
			international_organization:kc_kitayama_concord ?= { has_variable = kc_outer_port_charters_unlocked }
			OR = {
				this = location:hakata
				this = location:nishisonogi
				this = location:kagoshima
				this = location:tsuruga
				this = location:nutari
				this = location:toshima_kanto
				this = location:kameda
			}
		}
		# The second authorization follows the southern and Pacific coast from
		# Nishimuro around the Kii peninsula and eastward through Kashima. An
		# explicit list avoids granting the building to inland province members.
		AND = {
			international_organization:kc_kitayama_concord ?= { has_variable = kc_southern_ocean_harbor_line_unlocked }
			OR = {
				this = location:nishimuro
				this = location:minamimuro
				this = location:higashimuro

				this = location:kamo_izu
