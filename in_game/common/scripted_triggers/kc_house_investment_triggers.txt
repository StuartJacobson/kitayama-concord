# Event root and scope:actor are the investing House. Current scope is a location.
kc_house_can_order_overseas_building = {
	# Avoid piling scripted orders into a busy port or an existing build queue.
	num_civil_constructions = 0
	can_build_building = building_type:$building$
	root = {
		can_build_building = building_type:$building$
		# A conservative base-price reserve; normal native construction still
		# calculates and charges the real price. No free buildings or grants.
		gold >= {
			value = building_type:$building$.building_base_cost_in_gold
			multiply = 2
			add = { value = monthly_income_trade_and_tax multiply = 3 }
			min = 100
		}
	}
}
