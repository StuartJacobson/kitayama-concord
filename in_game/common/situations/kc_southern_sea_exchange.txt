kc_southern_sea_exchange = {
	monthly_spawn_chance = 0

	can_start = { always = no }

	can_end = {
		custom_tooltip = {
			text = kc_southern_sea_exchange_end_tt
			OR = {
				var:kc_sse_progress >= 100
				NOT = { country_exists = c:JAP }
				NOT = { exists = international_organization:kc_kitayama_concord }
				c:JAP ?= {
					OR = {
						NOT = { has_reform = government_reform:kc_kitayama_concordate }
						has_variable = kc_concord_collapsed
					}
				}
			}
		}
	}

	visible = {
		OR = {
			tag = JAP
			is_member_of_international_organization = international_organization:kc_kitayama_concord
		}
	}

	on_start = {
		set_variable = { name = kc_sse_progress value = 0 }
		set_variable = { name = kc_sse_monthly_progress value = 0.35 }
		international_organization:kc_kitayama_concord ?= {
			set_variable = { name = kc_southern_sea_exchange_active value = yes }
		}
		c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.1 }
	}

	on_monthly = {
		custom_tooltip = kc_southern_sea_exchange_monthly_tt

		hidden_effect = {
			set_variable = { name = kc_sse_monthly_progress value = 0.35 }

			# Full salaries for interpreters, pilots, and maritime judges are the
			# single largest accelerator. Native bureaucracy Entrenchment also
			# strengthens this custom funding value.
			if = {
				limit = {
					c:JAP ?= {
						has_bureaucracy_of_type = bureaucracy_type:kc_office_maritime_exchange_bureaucracy
						modifier:kc_maritime_office_underfunding <= 0
					}
				}
				change_variable = { name = kc_sse_monthly_progress add = 0.25 }
			}
			else_if = {
				limit = {
					c:JAP ?= {
						has_bureaucracy_of_type = bureaucracy_type:kc_office_maritime_exchange_bureaucracy
						modifier:kc_maritime_office_funding > 0
					}
				}
				change_variable = { name = kc_sse_monthly_progress add = 0.10 }
			}

			if = {
				limit = { c:JAP ?= { has_advance = kc_licensed_oceanic_convoys } }
				change_variable = { name = kc_sse_monthly_progress add = 0.10 }
			}

			if = {
				limit = {
					c:JAP ?= {
						any_owned_location = {
							region = region:japan_region
							is_coastal = yes
							is_city = yes
						}
					}
				}
				change_variable = { name = kc_sse_monthly_progress add = 0.05 }
			}

			if = {
				limit = {
					c:JAP ?= {
						OR = {
							has_reform = government_reform:kc_council_of_licensed_ports
							has_reform = government_reform:kc_national_board_of_exchange
						}
					}
				}
				change_variable = { name = kc_sse_monthly_progress add = 0.05 }
			}

			if = {
				limit = { c:JAP ?= { at_war = yes } }
				change_variable = { name = kc_sse_monthly_progress add = -0.15 }
			}

			change_variable = { name = kc_sse_progress add = var:kc_sse_monthly_progress }
			clamp_variable = { name = kc_sse_progress min = 0 max = 100 }

			if = {
				limit = { var:kc_sse_progress >= 20 NOT = { has_variable = kc_sse_arms_ratification } }
				set_variable = { name = kc_sse_arms_ratification value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.2 }
			}
			if = {
				limit = { var:kc_sse_progress >= 40 NOT = { has_variable = kc_sse_powder_incident } }
				set_variable = { name = kc_sse_powder_incident value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.3 }
			}
			if = {
				limit = { var:kc_sse_progress >= 55 NOT = { has_variable = kc_sse_silver_ratification } }
				set_variable = { name = kc_sse_silver_ratification value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.4 }
			}
			if = {
				limit = { var:kc_sse_progress >= 70 NOT = { has_variable = kc_sse_translation_ratification } }
				set_variable = { name = kc_sse_translation_ratification value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.5 }
			}
			if = {
				limit = { var:kc_sse_progress >= 76 NOT = { has_variable = kc_sse_southern_islands_survey } }
				set_variable = { name = kc_sse_southern_islands_survey value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_islands.1 }
			}
			if = {
				limit = { var:kc_sse_progress >= 82 NOT = { has_variable = kc_sse_missionary_incident } }
				set_variable = { name = kc_sse_missionary_incident value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.6 }
			}
			if = {
				limit = { var:kc_sse_progress >= 92 NOT = { has_variable = kc_sse_port_ratification } }
				set_variable = { name = kc_sse_port_ratification value = yes }
				c:JAP ?= { trigger_event_non_silently = kc_southern_exchange.7 }
			}
		}
	}

	on_ending = {
		if = {
			limit = {
				var:kc_sse_progress >= 100
				country_exists = c:JAP
				exists = international_organization:kc_kitayama_concord
				c:JAP ?= {
					has_reform = government_reform:kc_kitayama_concordate
					NOT = { has_variable = kc_concord_collapsed }
				}
			}
			international_organization:kc_kitayama_concord ?= {
				remove_variable = kc_southern_sea_exchange_active
			}
			c:JAP ?= {
				set_variable = { name = kc_southern_sea_exchange_completed value = yes }
				remove_variable = kc_historical_chain_active
				trigger_event_non_silently = kc_southern_exchange.10
			}
		}
		else = {
			international_organization:kc_kitayama_concord ?= { remove_variable = kc_southern_sea_exchange_active }
			c:JAP ?= {
				remove_variable = kc_historical_chain_active
				trigger_event_non_silently = kc_southern_exchange.11
			}
		}
	}

	tooltip = {
		if = {
			limit = { owner ?= { tag = JAP } }
			custom_tooltip = kc_southern_sea_exchange_map_state_tt
		}
		else_if = {
			limit = { owner ?= { is_member_of_international_organization = international_organization:kc_kitayama_concord } }
			custom_tooltip = kc_southern_sea_exchange_map_house_tt
		}
	}

	map_color = {
		if = { limit = { owner ?= { tag = JAP } } value = define:NMapColors|MAP_COLOR_HIGH }
		else_if = { limit = { owner ?= { is_member_of_international_organization = international_organization:kc_kitayama_concord } } value = define:NMapColors|MAP_COLOR_MID }
		else = { value = define:NMapColors|DEFAULT_COLOR }
	}

	legend_key = { desc = "KC_SSE_LEGEND_STATE" color = define:NMapColors|MAP_COLOR_HIGH }
	legend_key = { desc = "KC_SSE_LEGEND_HOUSES" color = define:NMapColors|MAP_COLOR_MID }
}
