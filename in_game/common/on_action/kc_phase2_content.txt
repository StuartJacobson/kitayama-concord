# One-time save-compatible backfill. New Concords receive the relations during
# formation; older saves receive them on their first monthly country pulse.
kc_concession_rights_monthly_pulse = {
	trigger = {
		tag = JAP
		has_reform = government_reform:kc_kitayama_concordate
		exists = international_organization:kc_kitayama_concord
		OR = {
			NOT = { has_variable = kc_concession_rights_initialized }
			any_subject = { NOT = { has_variable = kc_concession_rights_granted } }
		}
	}
	effect = { kc_refresh_concord_concession_rights = yes }
}

# Keep a rolling ownership ledger while Japan administers a colonial province,
# then repair any recorded House concessions after that province is assigned to
# a colonial nation. This covers both vanilla transfer options and delayed
# ownership cleanup which can outlive on_location_changed_owner by a tick.
kc_colonial_concession_persistence_monthly_pulse = {
	trigger = {
		tag = JAP
		has_reform = government_reform:kc_kitayama_concordate
		exists = international_organization:kc_kitayama_concord
	}
	effect = {
		# Snapshot only locations which actually contain one of the colonial
		# House concessions, avoiding a full cache write across the empire.
		every_owned_location = {
			limit = {
				OR = {
					continent = continent:america
					continent = continent:oceania
				}
				OR = {
					any_foreign_buildings_in_location = { building_type = building_type:kc_house_anchorage }
					any_foreign_buildings_in_location = { building_type = building_type:kc_colonial_house_exchange }
					any_foreign_buildings_in_location = { building_type = building_type:kc_colonial_tea_gardens }
					any_foreign_buildings_in_location = { building_type = building_type:kc_colonial_silk_farm }
					any_foreign_buildings_in_location = { building_type = building_type:kc_chartered_paddy_syndicate }
					any_foreign_buildings_in_location = { building_type = building_type:kc_house_alum_works }
					any_foreign_buildings_in_location = { building_type = building_type:kc_northern_house_whaling_station }
					any_foreign_buildings_in_location = { building_type = building_type:kc_house_provisioning_depot }
					any_foreign_buildings_in_location = { building_type = building_type:kc_chartered_timber_yard }
				}
			}
			kc_cache_colonial_concessions_in_location = yes
		}

		every_subject = {
			limit = { is_subject_type = colonial_nation }
			kc_grant_concord_concession_rights_to_houses = yes
			every_owned_location = {
				limit = {
					OR = {
						has_variable = kc_cached_anchorage_levels
						has_variable = kc_cached_exchange_levels
						has_variable = kc_cached_tea_levels
						has_variable = kc_cached_silk_levels
						has_variable = kc_cached_paddy_levels
						has_variable = kc_cached_alum_levels
						has_variable = kc_cached_whaling_levels
						has_variable = kc_cached_provisioning_levels
						has_variable = kc_cached_timber_levels
					}
				}
				kc_restore_colonial_concessions_in_location = yes
			}
		}
	}
}

# Save-compatible bridge for campaigns that completed Regulation of the Open
# Seas before the late Harbor authorizations were introduced.
kc_late_maritime_works_backfill_monthly_pulse = {
	trigger = {
		tag = JAP
		has_variable = kc_regulation_open_seas_completed
		exists = international_organization:kc_kitayama_concord
		international_organization:kc_kitayama_concord ?= {
			NOT = { has_variable = kc_late_maritime_works_available }
			NOT = { has_variable = kc_concord_collapsed }
		}
	}
	effect = {
		international_organization:kc_kitayama_concord = {
			set_variable = { name = kc_late_maritime_works_available value = yes }
		}
	}
}

kc_taiwan_settlement_monthly_pulse = {
	trigger = {
		tag = JAP
		has_reform = government_reform:kc_kitayama_concordate
		has_variable = kc_southern_sea_exchange_completed
		area:taiwan_area = {
			any_province_in_area = {
				owner ?= root
				any_location_in_province = {
					OR = { has_owner = no AND = { has_owner = yes NOT = { owner = root } } }
					count = 0
				}
				NOT = { has_variable = kc_taiwan_settlement_policy }
				NOT = { has_variable = kc_taiwan_settlement_event_pending }
			}
		}
	}
	effect = {
		area:taiwan_area = {
			random_province_in_area = {
				limit = {
					owner ?= root
					any_location_in_province = {
						OR = { has_owner = no AND = { has_owner = yes NOT = { owner = root } } }
						count = 0
					}
					NOT = { has_variable = kc_taiwan_settlement_policy }
					NOT = { has_variable = kc_taiwan_settlement_event_pending }
				}
				set_variable = { name = kc_taiwan_settlement_event_pending value = yes }
			}
		}
		trigger_event_non_silently = kc_overseas_settlements.1
	}
}

# A Japanese foothold can be transferred to the established harbor polity in
# return for a negotiated Commercial Protectorate compact.
kc_philippine_compacts_monthly_pulse = {
	trigger = {
		tag = JAP
		has_variable = kc_southern_sea_exchange_completed
		NOT = { has_variable = kc_historical_chain_active }
		OR = {
			AND = {
				exists = c:TDO
				NOT = { has_variable = kc_luzon_compact_completed }
				NOT = { has_variable = kc_luzon_compact_declined }
				OR = {
					any_owned_location = { area = area:taiwan_area }
					location:maynila.market.most_powerful_merchant = root
					any_owned_location = { area = area:luzon_area }
				}
				c:TDO = { is_subject = no }
			}
			AND = {
				exists = c:CEB
				NOT = { has_variable = kc_visayas_compact_completed }
				NOT = { has_variable = kc_visayas_compact_declined }
				OR = {
					any_owned_location = { area = area:taiwan_area }
					location:maynila.market.most_powerful_merchant = root
					any_owned_location = { area = area:visayas_area }
				}
				c:CEB = { is_subject = no }
			}
			AND = {
				exists = c:BTU
				NOT = { has_variable = kc_mindanao_compact_completed }
				NOT = { has_variable = kc_mindanao_compact_declined }
				OR = {
					any_owned_location = { area = area:taiwan_area }
					location:maynila.market.most_powerful_merchant = root
					any_owned_location = { area = area:mindanao_area }
				}
				c:BTU = { is_subject = no }
			}
		}
	}
	effect = {
		if = {
			limit = { exists = c:TDO NOT = { has_variable = kc_luzon_compact_completed } NOT = { has_variable = kc_luzon_compact_declined } OR = { any_owned_location = { area = area:taiwan_area } location:maynila.market.most_powerful_merchant = root any_owned_location = { area = area:luzon_area } } c:TDO = { is_subject = no } }
			trigger_event_non_silently = kc_overseas_settlements.2
		}
		else_if = {
			limit = { exists = c:CEB NOT = { has_variable = kc_visayas_compact_completed } NOT = { has_variable = kc_visayas_compact_declined } OR = { any_owned_location = { area = area:taiwan_area } location:maynila.market.most_powerful_merchant = root any_owned_location = { area = area:visayas_area } } c:CEB = { is_subject = no } }
			trigger_event_non_silently = kc_overseas_settlements.3
		}
		else = { trigger_event_non_silently = kc_overseas_settlements.4 }
	}
}

# The survey claim is a deliberately finite opening. Completed settlements
# remain Japanese, but unused exclusive rights lapse after thirty years.
kc_southern_island_access_expiry_monthly_pulse = {
	trigger = {
		tag = JAP
		has_variable = kc_southern_islands_survey_completed
		OR = {
			AND = {
				has_variable = kc_southern_islands_access_window
				NOT = { has_variable = kc_southern_islands_cleanup_scheduled }
			}
			AND = {
				NOT = { has_variable = kc_southern_islands_access_window }
				NOT = { has_variable = kc_southern_islands_access_expired }
			}
		}
	}
	effect = {
		if = {
			limit = { has_variable = kc_southern_islands_access_window }
			# Alpha.54 saves with a persistent access variable receive a fresh,
			# guaranteed cleanup schedule rather than retaining claims forever.
			set_variable = { name = kc_southern_islands_cleanup_scheduled value = yes }
			trigger_event_silently = { id = kc_southern_islands.2 days = 10950 }
		}
		else = { kc_expire_southern_islands_access = yes }
	}
}

# New qualifying subjects receive the overseas-community charter once. This
# iterates subjects rather than locations and never removes or reapplies the
# modifier from already registered capitals.
kc_overseas_communities_backfill_monthly_pulse = {
	trigger = {
		tag = JAP
		has_variable = kc_overseas_communities_completed
		any_subject = {
			OR = {
				is_subject_type = colonial_nation
				is_subject_type = kc_commercial_protectorate
				is_subject_type = kc_chartered_extraction_commission
			}
			NOT = { has_variable = kc_registered_nihonmachi_received }
		}
	}
	effect = {
		every_subject = {
			limit = {
				OR = {
					is_subject_type = colonial_nation
					is_subject_type = kc_commercial_protectorate
					is_subject_type = kc_chartered_extraction_commission
				}
				NOT = { has_variable = kc_registered_nihonmachi_received }
			}
			capital = { add_location_modifier = { modifier = kc_registered_nihonmachi years = -1 mode = replace } }
			set_variable = { name = kc_registered_nihonmachi_received value = yes }
		}
	}
}

# The final reform has a modest permanent core. Its more specialized benefits
# require a meaningfully funded Office of Maritime Exchange and expire quickly
# after its maintenance is reduced.

		tag = JAP

		has_bureaucracy_of_type = bureaucracy_type:kc_office_maritime_exchange_bureaucracy

		if = {
			limit = { has_reform = government_reform:kc_commonwealth_chartered_ports }
			add_country_modifier = { modifier = kc_chartered_ports_funded_office months = 2 mode = replace }
		}

			limit = { has_reform = government_reform:kc_articles_registered_open_ports }
			add_country_modifier = { modifier = kc_registered_ports_funded_office months = 2 mode = replace }
		}
		else_if = {
			limit = { has_reform = government_reform:kc_national_maritime_administration }
			add_country_modifier = { modifier = kc_national_maritime_funded_office months = 2 mode = replace }
		}
	}
}

# Chartered Houses are building-country actors and therefore possess no
# territorial population from which the native AI can derive a positive fleet
# target. Even a wealthy House with sailors and a working shipyard consequently
# evaluates its expected navy as zero. This AI-only, paid procurement pulse
# establishes a small law-sensitive baseline without changing player Houses.
kc_house_naval_procurement_yearly_pulse = {
	trigger = {
		is_ai = yes
		has_reform = government_reform:kc_chartered_buke_house
		is_member_of_international_organization = international_organization:kc_kitayama_concord
		NOT = { has_variable = kc_house_ship_order_cooldown }
		sailors >= 0.10
		gold >= { value = monthly_income_trade_and_tax min = 10 max = 100 multiply = 6 }
		any_owned_foreign_building_location = {
			is_port = yes
			any_foreign_buildings_in_location = {
				building_type = building_type:clan_shipyards
				owner = root
			}
		}
		international_organization:kc_kitayama_concord = {
			NOT = { has_variable = kc_concord_collapsed }
			OR = {
				AND = {
					international_organization_has_policy = policy:kc_house_port_privileges_policy
					OR = {
						root = { navy_size < 1 }
						root = {
							navy_size < 3
							has_special_status_in_international_organization = {
								type = special_status:kc_senior_chartered_house
								international_organization = international_organization:kc_kitayama_concord
							}
						}
					}
				}
				AND = {
					international_organization_has_policy = policy:kc_licensed_coastal_convoys_policy
					root = {
						navy_size < 1
						has_special_status_in_international_organization = {
							type = special_status:kc_senior_chartered_house
							international_organization = international_organization:kc_kitayama_concord
						}
					}
				}
				AND = {
					international_organization_has_policy = policy:kc_concord_maritime_board_policy
					root = {
						navy_size < 1
						has_special_status_in_international_organization = {
							type = special_status:kc_senior_chartered_house
							international_organization = international_organization:kc_kitayama_concord
						}
					}
				}
			}
		}
	}

	effect = {
		save_scope_as = kc_shipbuilding_house
		add_gold = { value = monthly_income_trade_and_tax min = 10 max = 100 multiply = -6 }
		add_sailors = -0.10
		set_variable = { name = kc_house_ship_order_cooldown value = yes years = 2 }
		random_owned_foreign_building_location = {
			limit = {
				is_port = yes
				any_foreign_buildings_in_location = {
					building_type = building_type:clan_shipyards
					owner = scope:kc_shipbuilding_house
				}
			}
			save_scope_as = kc_house_shipyard_location
			random_list = {
				70 = { kc_spawn_house_galley_for_current_age = yes }
				30 = { kc_spawn_house_transport_for_current_age = yes }
			}
		}
		force_recalc_country_active_status = scope:kc_shipbuilding_house
	}
}
