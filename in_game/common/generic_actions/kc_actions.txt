# During The Fracturing Concord, JAP may subscribe six months of current
# trade-and-tax income to the common reserve. This is a real transfer rather
# than generated money and can be repeated after a six-month cooldown.

kc_emergency_treasury_subscription = {
	type = internationalorganization
	force_click_and_confirm_or_hold = yes
	show_in_gui_list = no
	show_message = no

	ai_tick = monthly
	ai_tick_frequency = 1
	automation_tick = never

	potential = {
		international_organization:kc_kitayama_concord ?= {
			NOT = { has_variable = kc_concord_collapsed }
		}
		scope:actor = {
			tag = JAP
			has_reform = government_reform:kc_kitayama_concordate
			is_member_of_international_organization = international_organization:kc_kitayama_concord
		}
		exists = international_organization:kc_kitayama_concord
		international_organization:kc_kitayama_concord ?= {
			has_variable = kc_fracturing_concord_active
		}
	}

	allow = {
		scope:recipient = {
			international_organization_type = international_organization_type:kc_kitayama_concord
			has_variable = kc_fracturing_concord_active
			custom_tooltip = {
				text = kc_emergency_subscription_ready_tt
				NOT = {
					has_variable = kc_emergency_subscription_cooldown
				}
			}
		}
		scope:actor = {
			custom_tooltip = {
				text = kc_emergency_subscription_funds_tt
				gold >= kc_treasury_six_month_cost
			}
		}
	}

	select_trigger = {
		looking_for_a = international_organization
		source = actor
		target_flag = recipient
		name = "choose_international_organization"
		none_available_msg_key = "trigger_no_kc_concord_available"
		column = { data = name }
		visible = {
			international_organization_type = international_organization_type:kc_kitayama_concord
		}
		enabled = {
			international_organization_type = international_organization_type:kc_kitayama_concord
		}
	}

	effect = {
		custom_tooltip = kc_emergency_treasury_subscription_effect_tt
			hidden_effect = {
			scope:actor = {
				add_gold = {
					value = kc_treasury_six_month_cost
					multiply = -1
				}
			}
			scope:recipient = {
				add_gold = {
					value = kc_treasury_six_month_cost
				}
				set_variable = {
					name = kc_emergency_subscription_cooldown
					value = yes
					months = 6
				}
			}
		}
	}

	ai_will_do = {
		value = -1000
		if = {
			limit = { scope:actor = { kc_ai_japan_active_concord = yes } scope:recipient = { gold < kc_treasury_six_month_cost } }
			add = 1100
		}
	}
}
