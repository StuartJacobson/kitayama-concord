# Ordinary colonial concessions for the Chartered Houses. Unlike the stronger
# buildings available inside Commercial Protectorates and Extraction
# Commissions, these require no concession license and impose no crown-power
# penalty. The first House to establish each scalable industry reserves that
# industry in the location and may develop all locally available levels.

kc_house_anchorage = {
	audio_tier = 2
	is_special = yes
	is_foreign = yes
	stronger_power_projection = no
	need_good_relation = no
	own_or_overlord_relation_needed = kc_concord_concession_rights
	want_foreign_pop_created = no
	pop_type = soldiers
	category = naval_category
	max_levels = 1
	employment_size = 0.20
	pop_size_created = 0.20
	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes
	ai_foreign_ignore_naval_range = yes
	AI_ignore_available_worker_flag = yes
	location_potential = {
		is_port = yes
		OR = {
			dominant_culture ?= { has_culture_group = culture_group:japanese_group }
			kc_japanese_colonial_house_location = yes
		}
		NOT = { any_foreign_buildings_in_location = { building_type = building_type:kc_house_anchorage } }
	}
	country_potential = { kc_chartered_house_building_country = yes }
	allow = {
		OR = {
			owner ?= { is_member_of_international_organization = international_organization:kc_kitayama_concord }
			kc_japanese_colonial_house_location = yes
		}
	}
	build_time = trade_post_build_time
	modifier = {
		gold_to_building_owner = 0.20
		sailors_to_building_owner = 0.003
		harbor_suitability = 0.05
	}
	possible_production_methods = { early_soldier_building_maintenance }
	construction_demand = port_building_construction
}
