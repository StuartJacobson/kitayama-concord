# Common Works no longer chooses five opaque random locations. Each Treasury
# commission instead issues five construction licenses to JAP. Concord
# locations remain JAP-owned even when a Chartered House operates there, so a
# single domestic type covers both rural and urban investment throughout the
# country.
kc_concord_works_district = {
	audio_tier = 2
	is_special = yes
	is_foreign = no
	pop_type = burghers

	category = infrastructure_category
	max_levels = 5
	employment_size = generic_burgher_employment
	increase_per_level_cost = 0.25

	rural_settlement = yes
	town = yes
	city = yes
	megalopolis = yes

	important_for_UI = yes

	location_potential = {
		owner ?= {
			tag = JAP
		}
	}

	country_potential = {
		tag = JAP
		has_reform = government_reform:kc_kitayama_concordate
		is_member_of_international_organization = international_organization:kc_kitayama_concord
		international_organization:kc_kitayama_concord ?= {
			has_variable = kc_common_works_licenses_available
		}
	}

	allow = {
		scope:actor = {
			tag = JAP
			has_reform = government_reform:kc_kitayama_concordate
			is_member_of_international_organization = international_organization:kc_kitayama_concord
		}
		owner ?= {
			tag = JAP
		}
		international_organization:kc_kitayama_concord ?= {
			has_variable = kc_common_works_licenses_available
		}
	}

	build_time = generic_town_build_time

	modifier = {
		free_building_levels = 11
	}

	unique_production_methods = {
		kc_concord_works_district_maintenance = {
			lumber = 0.10
			stone = 0.05
			tools = 0.05
			category = building_maintenance
		}
	}

	on_construction_started = {
		international_organization:kc_kitayama_concord ?= {
			change_variable = {
				name = kc_common_works_licenses
				add = -1
			}
			if = {
				limit = {
					var:kc_common_works_licenses < 1
				}
				remove_variable = kc_common_works_licenses_available
			}
		}
	}

	# The generic construction-ended hook also runs after cancellation. Restore
	# the reservation here, then consume it again only if a level was completed.
	on_construction_ended = {
		international_organization:kc_kitayama_concord ?= {
			change_variable = {
				name = kc_common_works_licenses
				add = 1
			}
			set_variable = {
				name = kc_common_works_licenses_available
				value = yes
			}
		}
	}

	on_built = {
		international_organization:kc_kitayama_concord ?= {
			change_variable = {
				name = kc_common_works_licenses
				add = -1
			}
			if = {
				limit = {
					var:kc_common_works_licenses < 1
				}
				remove_variable = kc_common_works_licenses_available
			}
		}
	}

	construction_demand = early_capital_building_construction
}
