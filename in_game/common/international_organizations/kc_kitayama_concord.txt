	leader_type = character
	leader = {
		leader_country ?= {
			if = {
				limit = {
					NOT = {
						is_member_of_international_organization = root
					}
				}
				root = {
					add_country_to_international_organization = prev
				}
			}

			if = {
				limit = { has_regent = yes }
				heir ?= {
					add_to_list = leaders
				}
			}
			else = {
				ruler ?= {
					add_to_list = leaders
				}
			}
		}
	}

	leader_change_trigger_type = rulerchange

	can_lead_trigger = {
		trigger_if = {
			limit = {

					has_variable = kc_concord_collapsed
				}
			}
			tag = JAP

		trigger_else = {
			country_type = building
			has_ruler = yes
			OR = {
				has_reform = government_reform:kc_chartered_buke_house
				has_reform = government_reform:kc_kitayama_concordate

			NOT = {
				has_special_status_in_international_organization = {
					type = special_status:japanese_emperor
					international_organization = international_organization:kc_kitayama_concord
				}
			}
		}
	}

		scope:actor ?= { NOT = { tag = JAP } }
	}

	join_defensive_wars_can_call = {

		scope:actor ?= { tag = JAP }
		scope:recipient ?= { has_reform = government_reform:kc_chartered_buke_house }
		international_organization_has_policy = policy:kc_discretionary_house_service_policy
	}

	join_defensive_wars_always = {

					is_member_of_international_organization = scope:recipient
				}

				limit = {
					scope:recipient = {
						has_variable = kc_concord_collapsed
					}

			}
			trigger_else = {
				custom_tooltip = {

		}
		trigger_else = {

	}

	on_creation = {
	}

	on_joined = {
		if = {
			limit = {
				country_type = building
				NOT = {
					has_reform = government_reform:japanese_imperial_family
				}
			}
			if = {
				limit = {
					has_reform = government_reform:japanese_clan
				}
				remove_reform = government_reform:japanese_clan
			}
			if = {
				limit = {
					has_reform = government_reform:shogunate
				}
				remove_reform = government_reform:shogunate
			}
			if = {
				limit = {
					tag = JAP
				}
				if = {
					limit = {
						has_reform = government_reform:kc_chartered_buke_house
					}
					remove_reform = government_reform:kc_chartered_buke_house
				}
				if = {
					limit = {
						NOT = {
							has_reform = government_reform:kc_kitayama_concordate
						}
					}
					add_reform = government_reform:kc_kitayama_concordate
				}
				change_country_name = KC_JAPAN_COUNTRY
				change_country_adjective = KC_JAPAN_COUNTRY_ADJ
			}
			else = {
				if = {
					limit = {
						NOT = {
							has_reform = government_reform:kc_chartered_buke_house
						}
					}
					add_reform = government_reform:kc_chartered_buke_house
				}
			}
		}
	}

	on_left = {
		if = {
			limit = {
				has_reform = government_reform:kc_chartered_buke_house
			}
			remove_reform = government_reform:kc_chartered_buke_house
		}
	}

	monthly_effect = {
		custom_tooltip = { text = kc_concord_monthly_maintenance_tt }
		hidden_effect = {
		if = {
			limit = {
				NOT = {
					has_variable = kc_concord_collapsed
				}
			}
		# Native ruler-change elections retain the first opportunity to fill a
		# vacancy. If the engine leaves the Concord leaderless for three monthly
		# pulses, choose the richest eligible member as a bounded fallback.
		# This never annexes or transfers the player country.
		if = {
			limit = {
				international_organization_has_leader = no
				NOT = {
					has_variable = kc_concord_vacancy_observed
				}
			}
			set_variable = {
				name = kc_concord_vacancy_observed
				value = yes
			}
			set_variable = {
				name = kc_concord_vacancy_grace
				value = yes
				months = 3
			}
		}
		else_if = {
			limit = {
				international_organization_has_leader = no
				has_variable = kc_concord_vacancy_observed
				NOT = {
					has_variable = kc_concord_vacancy_grace
				}
				any_international_organization_member = {
					can_lead_international_organization = root
				}
			}
			update_leader_country = yes
			remove_variable = kc_concord_vacancy_observed
		}
		else_if = {
			limit = {
				international_organization_has_leader = yes
				has_variable = kc_concord_vacancy_observed
			}
			remove_variable = kc_concord_vacancy_observed
			remove_variable = kc_concord_vacancy_grace
		}
		if = {
			limit = {
				NOT = {
					has_variable = imperial_authority
				}
			}
			set_variable = {
				name = imperial_authority
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = kc_last_annual_assessment
				}
			}
			set_variable = {
				name = kc_last_annual_assessment
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = kc_last_assessment_arrears
				}
			}
			set_variable = {
				name = kc_last_assessment_arrears
				value = 0
			}
		}

		# A funded Secretariat returns part of its nominal maintenance budget to
		# the common reserve. The current country modifier contains maintenance
		# and bureaucracy Entrenchment: 25% at zero Entrenchment, rising linearly
		# to 50% at 100. The remainder represents salaries, offices, couriers, and
		# other expenditure that never enters the discretionary Treasury.
		# EU5 permits a small floating overshoot after native Entrenchment reaches
		# 100, so normalize this bureaucracy before calculating its remittance.
		if = {
			limit = { country_exists = c:JAP }
			c:JAP = {
				every_current_bureaucracy = {
					limit = {
						OR = {
							bureaucracy_type = bureaucracy_type:kc_concord_secretariat_bureaucracy
							bureaucracy_type = bureaucracy_type:kc_office_maritime_exchange_bureaucracy
						}
						entrenchment > 100
					}
					set_entrenchment = 100
				}
			}
		}
		set_variable = { name = kc_secretariat_treasury_share_display value = 0 }
		set_variable = { name = kc_last_secretariat_remittance value = 0 }
		set_variable = { name = kc_secretariat_annual_remittance_estimate value = 0 }
		if = {
			limit = {
				country_exists = c:JAP
				c:JAP ?= {
					has_bureaucracy_of_type = bureaucracy_type:kc_concord_secretariat_bureaucracy
					modifier:kc_concord_secretariat_treasury_share > 0
				}
			}
			set_variable = {
				name = kc_secretariat_treasury_share_display
				value = {
					value = kc_secretariat_current_treasury_share
					multiply = 100
				}
			}
			set_variable = {
				name = kc_last_secretariat_remittance
				value = kc_secretariat_monthly_treasury_remittance
			}
			set_variable = {
				name = kc_secretariat_annual_remittance_estimate
				value = {
					value = kc_secretariat_monthly_treasury_remittance
					multiply = 12
				}
			}
			add_gold = kc_secretariat_monthly_treasury_remittance
		}

		# Vanilla Middle Kingdom membership admits a Shogunate member only if
		# it leads the Shogunate. Existing Concord saves may already contain
		# dozens of House tributaries from the missing equivalent restriction;
		# remove those subordinate memberships once, while leaving JAP free to
		# conduct centralized tally-trade diplomacy.
		if = {
			limit = { exists = international_organization:middle_kingdom }
			# Bound the migration to twelve Houses per month; bulk mutation of a
			# hundred-plus-member native organization is needlessly risky.
			ordered_international_organization_member = {
				limit = {
					has_reform = government_reform:kc_chartered_buke_house
					is_member_of_international_organization = international_organization:middle_kingdom
				}
				order_by = country_tax_base
				max = 12
				check_range_bounds = no
				save_scope_as = kc_middle_kingdom_house
				international_organization:middle_kingdom = {
					remove_country_from_international_organization = scope:kc_middle_kingdom_house
				}
			}
		}

		# Stabilize House Credit uses the strongest House tax base as its
		# benchmark instead of Japan's much larger national trade-and-tax income.
		set_variable = {
			name = kc_richest_house_tax_base
			value = 1
		}
		ordered_international_organization_member = {
			limit = {
				has_reform = government_reform:kc_chartered_buke_house
			}
			order_by = country_tax_base
			max = 1
			check_range_bounds = no
			save_scope_as = kc_richest_house
			root = {
				set_variable = {
					name = kc_richest_house_tax_base
					value = scope:kc_richest_house.country_tax_base
				}
			}
		}

		# Great-project stages are explicit IO variables so the Treasury ledger,
		# dynamic costs, and save files always agree. Missing variables from old
		# saves begin at zero.
		# EU5 discards numeric variables set to exact zero. A non-visible epsilon
		# preserves the stage scope while still formatting and behaving as Stage 0.
		if = { limit = { NOT = { has_variable = kc_project_irrigation_stage } } set_variable = { name = kc_project_irrigation_stage value = 0.0001 } }
		if = { limit = { NOT = { has_variable = kc_project_kaido_stage } } set_variable = { name = kc_project_kaido_stage value = 0.0001 } }
		if = { limit = { NOT = { has_variable = kc_project_harbors_stage } } set_variable = { name = kc_project_harbors_stage value = 0.0001 } }
		if = { limit = { NOT = { has_variable = kc_project_floodworks_stage } } set_variable = { name = kc_project_floodworks_stage value = 0.0001 } }
		if = { limit = { NOT = { has_variable = kc_project_tunnels_stage } } set_variable = { name = kc_project_tunnels_stage value = 0.0001 } }
		if = { limit = { NOT = { has_variable = kc_project_waterworks_stage } } set_variable = { name = kc_project_waterworks_stage value = 0.0001 } }

		# Alpha.20/21 used one-click completion flags and full-strength modifiers.
		# Preserve those investments as stage I rather than discarding them.
		if = {
			limit = { has_variable = kc_project_irrigation_completed }
			remove_variable = kc_project_irrigation_completed
			set_variable = { name = kc_project_irrigation_stage value = 1 }
			c:JAP ?= {
				remove_country_modifier = kc_project_grand_irrigation
				add_country_modifier = { modifier = kc_project_grand_irrigation_1 years = -1 mode = replace }
			}
		}
		if = {
			limit = { has_variable = kc_project_kaido_completed }
			remove_variable = kc_project_kaido_completed
			set_variable = { name = kc_project_kaido_stage value = 1 }
			c:JAP ?= {
				remove_country_modifier = kc_project_kaido_spine
				add_country_modifier = { modifier = kc_project_kaido_spine_1 years = -1 mode = replace }
			}
		}
		if = {
			limit = { has_variable = kc_project_harbors_completed }
			remove_variable = kc_project_harbors_completed
			set_variable = { name = kc_project_harbors_stage value = 1 }
			c:JAP ?= {
				remove_country_modifier = kc_project_inland_sea_harbors
				add_country_modifier = { modifier = kc_project_inland_sea_harbors_1 years = -1 mode = replace }
			}
		}
		if = {
			limit = { has_variable = kc_project_floodworks_completed }
			remove_variable = kc_project_floodworks_completed
			set_variable = { name = kc_project_floodworks_stage value = 1 }
			c:JAP ?= {
				remove_country_modifier = kc_project_floodworks
				add_country_modifier = { modifier = kc_project_floodworks_1 years = -1 mode = replace }
			}
		}
		if = {
			limit = { has_variable = kc_project_tunnels_completed }
			remove_variable = kc_project_tunnels_completed
			set_variable = { name = kc_project_tunnels_stage value = 1 }
			c:JAP ?= {
				remove_country_modifier = kc_project_mountain_tunnels
				add_country_modifier = { modifier = kc_project_mountain_tunnels_1 years = -1 mode = replace }
			}
		}

		# Alpha.24 localizes the scalable economic effects to the Japan region.
		# Rebuild the appropriate province/location tier once for existing saves.
		if = {
			limit = { NOT = { has_variable = kc_projects_localized_v1 } }
			set_variable = { name = kc_projects_localized_v1 value = yes }
			c:JAP ?= {
				remove_country_modifier = kc_project_grand_irrigation_1
				remove_country_modifier = kc_project_grand_irrigation_2
				remove_country_modifier = kc_project_grand_irrigation_3
				remove_country_modifier = kc_project_grand_irrigation_4
				remove_country_modifier = kc_project_grand_irrigation_5
				remove_country_modifier = kc_project_floodworks_1
				remove_country_modifier = kc_project_floodworks_2
				remove_country_modifier = kc_project_floodworks_3
				remove_country_modifier = kc_project_floodworks_4
				remove_country_modifier = kc_project_floodworks_5
				remove_country_modifier = kc_project_urban_waterworks_1
				remove_country_modifier = kc_project_urban_waterworks_2
				remove_country_modifier = kc_project_urban_waterworks_3
				remove_country_modifier = kc_project_urban_waterworks_4
				remove_country_modifier = kc_project_urban_waterworks_5
			}
			region:japan_region = {
				every_area_in_region = {
