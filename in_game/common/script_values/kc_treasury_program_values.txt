# All values are evaluated from the Kitayama Concord IO scope. Basing recurring
# programs on JAP's current trade-and-tax income keeps the common reserve
# relevant as Japan grows without creating an exponential late-game sink.

# The Secretariat's full nominal upkeep is five base gold multiplied by
# 0.008 of Japan's economic base, or 0.04 economic base. Its custom country
# modifier already contains both the maintenance setting and the native
# Entrenchment-scaled Treasury share (25% at zero, 50% at 100). Native
# bureaucracy Entrenchment can overshoot 100 by a floating monthly increment,
# so the derived share has its own hard ceiling as a second line of defense.
kc_secretariat_current_treasury_share = {
	value = c:JAP.modifier:kc_concord_secretariat_treasury_share
	min = 0
	max = 0.50
}

kc_secretariat_monthly_treasury_remittance = {
	value = c:JAP.country_economical_base
	multiply = 0.04
	multiply = kc_secretariat_current_treasury_share
	min = 0
}

# network building. These are evaluated from location scope by max_levels.

		limit = {
			exists = international_organization:kc_kitayama_concord
			international_organization:kc_kitayama_concord ?= { has_variable = kc_project_irrigation_stage }
		}
		add = international_organization:kc_kitayama_concord.var:kc_project_irrigation_stage

	min = 0
	max = 5
