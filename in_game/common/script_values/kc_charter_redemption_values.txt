# base of the richest current Chartered House. The backstop begins at twice
# that benchmark, adds half a benchmark for every House in the latest arrears
# count, and caps both its multiplier and final price.

	value = 2
	add = {
		value = var:kc_last_assessment_arrears
		multiply = 0.5
	}
	max = 12
}

	value = var:kc_richest_house_tax_base
	min = 1
	multiply = kc_stabilize_credit_months
