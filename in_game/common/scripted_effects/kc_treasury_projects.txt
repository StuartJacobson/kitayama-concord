# Preview the next tier of a five-stage Treasury project. The calling action
# supplies the IO variable and five country modifiers. This effect is used
# only inside show_as_tooltip, so the native modifier breakdown remains the
# authoritative description of an upgrade.
kc_preview_treasury_project_upgrade_effect = {
	if = {
		limit = { scope:recipient.var:$stage$ < 1 }
		add_country_modifier = { modifier = $tier_1$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 2 }
		add_country_modifier = { modifier = $tier_2$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 3 }
		add_country_modifier = { modifier = $tier_3$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 4 }
		add_country_modifier = { modifier = $tier_4$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 5 }
		add_country_modifier = { modifier = $tier_5$ years = -1 mode = replace }
	}
}

# Replace the current tier with the next one. The action deducts Treasury gold
# before calling this effect and increments the IO stage afterwards.
kc_apply_treasury_project_upgrade_effect = {
	if = {
		limit = { scope:recipient.var:$stage$ < 1 }
		add_country_modifier = { modifier = $tier_1$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 2 }
		remove_country_modifier = $tier_1$
		add_country_modifier = { modifier = $tier_2$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 3 }
		remove_country_modifier = $tier_2$
		add_country_modifier = { modifier = $tier_3$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 4 }
		remove_country_modifier = $tier_3$
		add_country_modifier = { modifier = $tier_4$ years = -1 mode = replace }
	}
	else_if = {
		limit = { scope:recipient.var:$stage$ < 5 }
		remove_country_modifier = $tier_4$
		add_country_modifier = { modifier = $tier_5$ years = -1 mode = replace }
	}
}

# Save migration helpers read the permanent Concord directly because their
# province/location scopes are nested beneath the IO's monthly effect.
kc_restore_concord_province_project_effect = {
	if = {
		limit = { international_organization:kc_kitayama_concord ?= { has_variable = $stage$ } }
		if = {
			limit = { international_organization:kc_kitayama_concord.var:$stage$ >= 5 }
			add_province_modifier = { modifier = $tier_5$ years = -1 mode = replace }
		}
		else_if = {
			limit = { international_organization:kc_kitayama_concord.var:$stage$ >= 4 }
			add_province_modifier = { modifier = $tier_4$ years = -1 mode = replace }
		}
		else_if = {
			limit = { international_organization:kc_kitayama_concord.var:$stage$ >= 3 }
			add_province_modifier = { modifier = $tier_3$ years = -1 mode = replace }
		}
		else_if = {
			limit = { international_organization:kc_kitayama_concord.var:$stage$ >= 2 }
			add_province_modifier = { modifier = $tier_2$ years = -1 mode = replace }
		}
		else_if = {

			add_province_modifier = { modifier = $tier_1$ years = -1 mode = replace }
		}
	}
}
