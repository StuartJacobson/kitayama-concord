kc_complete_ningbo_compact = {
	set_variable = { name = kc_ningbo_chain_completed value = yes }
	set_variable = { name = kc_phase2_convoys_unlocked value = yes }
	remove_variable = kc_historical_chain_active
	remove_variable = kc_historical_debug_rapid
	if = {
		limit = { NOT = { has_advance = kc_licensed_oceanic_convoys } }
		research_advance = advance_type:kc_licensed_oceanic_convoys
	}
}
