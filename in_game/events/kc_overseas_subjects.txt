kc_overseas_subjects.1 = { # A Harbor Petitions the Common Treasury
	type = country_event
	title = kc_overseas_subjects.1.title
	desc = {
		first_valid = {
			triggered_desc = { trigger = { scope:kc_overseas_subject = { has_variable = kc_korean_settlement_subject } } desc = kc_overseas_subjects.1.desc.korea }

	image = "gfx/interface/illustrations/situation/kc_southern_sea_exchange.dds"

		set_variable = { name = kc_overseas_subject_consequence_cooldown value = 1 years = 4 }
		set_variable = { name = kc_overseas_subject_1_recent value = 1 years = 16 }
		random_subject = { limit = { is_subject_type = kc_commercial_protectorate } save_scope_as = kc_overseas_subject }
	}
	option = {
		name = kc_overseas_subjects.1.a
		trigger = { international_organization:kc_kitayama_concord = { gold >= { value = kc_overseas_subject_minor_treasury_value } } }

		ai_chance = { base = 45 
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 0 }
		}

	option = {

		scope:kc_overseas_subject = { add_country_modifier = { modifier = kc_protectorate_house_harbor_concession years = 10 mode = replace } }
