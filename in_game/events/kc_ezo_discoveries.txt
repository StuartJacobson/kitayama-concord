namespace = kc_ezo_discoveries

kc_ezo_discoveries.1 = { # Black Sands of Shimamaki
	type = country_event
	title = kc_ezo_discoveries.1.title
	desc = kc_ezo_discoveries.1.desc
	outcome = positive
	image = "gfx/interface/icons/trade_goods/illustrations/icon_goods_iron.dds"
	trigger = {
		owns = location:shimamaki
		location:shimamaki = { raw_material = goods:sand }
		province:shiribeshi_province = { has_variable = kc_ezo_settlement_policy }
	}
	option = {
		name = kc_ezo_discoveries.1.a
		change_gold_effect = { scale = -1 }
		location:shimamaki = {
			change_raw_material = goods:iron
			add_location_modifier = { modifier = kc_shimamaki_ironsand_works years = 20 mode = replace }
		}
		change_societal_value = { type = traditionalist_vs_innovative value = 2 }
		custom_tooltip = {
			text = kc_concord_gain_1_cohesion_tt
			international_organization:kc_kitayama_concord = { change_variable = { name = kc_concord_cohesion add = 1 } }
		}
		ai_chance = { base = 60 
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 0 }
		}
	}
	option = {
		name = kc_ezo_discoveries.1.b
		location:shimamaki = { change_raw_material = goods:iron }
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		custom_tooltip = {
			text = kc_concord_lose_1_cohesion_tt
			international_organization:kc_kitayama_concord = { change_variable = { name = kc_concord_cohesion add = -1 } }
		}
		ai_chance = { base = 40 
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 10 }
		}
	}
}
