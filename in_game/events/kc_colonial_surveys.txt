kc_colonial_surveys.3 = {
	type = country_event
	title = kc_colonial_surveys.3.title
	desc = kc_colonial_surveys.3.desc
	outcome = positive
	image = "gfx/interface/illustrations/situation/kc_tallies_ningbo.dds"

	option = {
		name = kc_colonial_surveys.3.a
		custom_tooltip = kc_colonial_surveys_unlock_tt
		set_variable = { name = kc_phase2_colonial_surveys_unlocked value = yes }
		if = {
			limit = { NOT = { has_advance = kc_colonial_survey_registers } }
			research_advance = advance_type:kc_colonial_survey_registers
		}
		if = {
			limit = { has_variable = kc_colonial_common_register_chosen }
			add_country_modifier = { modifier = kc_colonial_common_register years = 20 mode = replace }
		}
		else_if = {
			limit = { has_variable = kc_colonial_chartered_registers_chosen }
			add_country_modifier = { modifier = kc_colonial_chartered_registers years = 20 mode = replace }
		}
		else_if = {
			limit = { has_variable = kc_colonial_treaty_compacts_chosen }
			add_country_modifier = { modifier = kc_colonial_treaty_compacts years = 20 mode = replace }
		}
		else = { add_country_modifier = { modifier = kc_colonial_common_register years = 20 mode = replace } }
		hidden_effect = {
			set_variable = { name = kc_colonial_surveys_completed value = yes }
			remove_variable = kc_colonial_surveys_debug_rapid
			remove_variable = kc_historical_chain_active
		}
	}
}
