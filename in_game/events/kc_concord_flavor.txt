namespace = kc_concord_flavor

# These events are selected by a dedicated yearly pulse. A two-year shared
# cooldown prevents event fatigue, while ten-year individual cooldowns keep
# the same story from recurring too often.

kc_concord_flavor.1 = { # Measures at the Inland Sea
	type = country_event
	title = kc_concord_flavor.1.title
	desc = kc_concord_flavor.1.desc
	outcome = neutral
	image = "gfx/interface/illustrations/international_organization_types/kc_kitayama_concord.dds"
	trigger = { NOT = { has_variable = kc_concord_flavor_1_recent } }
	immediate = {
		set_variable = { name = kc_concord_flavor_cooldown value = 1 years = 2 }
		set_variable = { name = kc_concord_flavor_1_recent value = 1 years = 10 }
	}
	option = {
		name = kc_concord_flavor.1.a
		change_gold_effect = { scale = -1 }
		add_country_modifier = { modifier = kc_concord_standard_measures years = 5 mode = replace }
		if = {
			limit = { has_variable = kc_concord_capstone_completed }
			change_societal_value = { type = house_privilege_vs_concord_commonwealth value = 2 }
		}
		else = { change_societal_value = { type = traditionalist_vs_innovative value = 2 } }
		custom_tooltip = {
			text = kc_concord_gain_1_cohesion_tt
			international_organization:kc_kitayama_concord = { change_variable = { name = kc_concord_cohesion add = 1 } }
		}
		ai_chance = { base = 60 
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 0 }
		}
	}
	option = {
		name = kc_concord_flavor.1.b
		add_estate_satisfaction = { type = estate_type:burghers_estate value = estate_satisfaction_mild_bonus }
		if = {
			limit = { has_variable = kc_concord_capstone_completed }
			change_societal_value = { type = house_privilege_vs_concord_commonwealth value = -2 }
		}
		custom_tooltip = {
			text = kc_concord_lose_2_cohesion_tt
			international_organization:kc_kitayama_concord = { change_variable = { name = kc_concord_cohesion add = -2 } }
		}
		ai_chance = { base = 40 
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 10 }
		}
	}
}
