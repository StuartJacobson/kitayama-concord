# event kc_debug.5
# Temporarily reduce the Concord to exactly 25 Chartered Houses so one final
# redemption can verify the permanent floor of 24. Run while paused; removed
# Houses will be discovered and re-added by the normal monthly membership
# repair after time resumes.
kc_debug.5 = {
	type = country_event
	orphan = yes
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		if = {
			limit = {
				has_variable = kc_charter_redemption_cooldown
			}
			remove_variable = kc_charter_redemption_cooldown
		}
		add_gold = 1000000

		international_organization:kc_kitayama_concord ?= {
			set_variable = {
				name = kc_concord_cohesion
				value = 100
			}
			set_variable = {
				name = kc_concord_entrenchment
				value = 100
			}
			while = {
				count = 200
				if = {
					limit = {
						any_international_organization_member = {
							has_reform = government_reform:kc_chartered_buke_house
							count > 25
						}
					}
					random_international_organization_member = {
						limit = {
							has_reform = government_reform:kc_chartered_buke_house
							is_ai = yes
							NOT = { tag = JAP }
							NOT = {
								is_leader_of_international_organization = international_organization:kc_kitayama_concord
							}
							NOT = {
								has_special_status_in_international_organization = {
									type = special_status:kc_senior_chartered_house
									international_organization = international_organization:kc_kitayama_concord
								}
							}
						}
						international_organization:kc_kitayama_concord = {
							remove_country_from_international_organization = prev
						}
					}
				}
			}
		}
	}
}

# member. Each previously unowned location receives one thousand sponsor-culture
# peasants so the three demographic policy choices can be compared directly.

		province_definition:oshima_province = {

				limit = {

				change_location_owner = root
				change_location_controller = root

					culture = root.culture

			}
		}
	}
}

# event kc_debug.13
# Construct an instant, zero-cost Kaisho in JAP's capital for 26 Chartered
# Houses that do not already operate a completed Kaido Post Station or Kaisho.
# This tests the real Registered House Watches building-owner requirement
# without using an override flag. Use only on a disposable or reloadable save.
kc_debug.13 = {
	type = country_event
	orphan = yes
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		international_organization:kc_kitayama_concord ?= {
			ordered_international_organization_member = {
				limit = {
					has_reform = government_reform:kc_chartered_buke_house
					NOT = {
						OR = {
							any_owned_foreign_building = {
								building_type = building_type:kc_kaido_post_station
								building_level > 0
							}
							any_owned_foreign_building = {
								building_type = building_type:kc_kaisho
								building_level > 0
							}
						}
					}
				}
				order_by = country_tax_base
				max = 26
				check_range_bounds = no
				save_scope_as = kc_debug_watch_house
				c:JAP = {
					capital ?= {
						construct_building = {
							building_type = building_type:kc_kaisho
							owner = scope:kc_debug_watch_house
							payer = c:JAP
							instant = yes
							cost_multiplier = 0
							cost_multiplier_reason = "game_concept_event"
						}
					}
				}
			}
		}
	}
}

# event kc_debug.15

# all Ledger Office levels, and ensure at least eight ordinary Houses hold a
# Senior Charter. This leaves only the route's Council policy to test.

	type = country_event
	orphan = yes
	outcome = neutral
	title = empty_text
	desc = empty_text
	hidden = yes

	immediate = {
		if = {

			name = kc_great_reckoning_ledger_office_authorized
			value = yes
		}
		capital ?= {
			change_building_level_in_location = {
				building = building_type:kc_concord_ledger_office

			set_variable = {
				name = kc_concord_cohesion
				value = 100
			}
			add_gold = {

				min = 1
				multiply = 15
			}
			ordered_international_organization_member = {
				limit = {
					has_reform = government_reform:kc_chartered_buke_house
					NOT = {
						has_special_status_in_international_organization = {
							type = special_status:kc_senior_chartered_house
							international_organization = international_organization:kc_kitayama_concord
						}
					}
				}
				order_by = country_tax_base

# event kc_debug.73 - one-time exploration backfill. Every current Concord
# member discovers each area already known to Japan. Ordinary gameplay uses the
# much cheaper on_exploration_success hook instead.
kc_debug.73 = {
	type = country_event
	orphan = yes
	outcome = neutral
	title = kc_debug.73.title
	desc = kc_debug.73.desc
	option = {
		name = kc_debug.73.a
		trigger = {
			tag = JAP
			exists = international_organization:kc_kitayama_concord
		}
		every_area = {
			limit = { root = { has_discovered_area = prev } }
			save_scope_as = kc_debug_shared_area
			international_organization:kc_kitayama_concord = {
				every_international_organization_member = {
					limit = {
						NOT = { has_discovered_area = scope:kc_debug_shared_area }
					}
					discover_area = scope:kc_debug_shared_area
				}
			}
		}
	}
}

			}
		}
	}
}

# event kc_debug.77
# Overseas House investment retest. Run as JAP, then advance several months.
# This clears the shared and personal scripted-AI construction throttles,
# refreshes construction rights, and gives AI Houses a modest test reserve.
kc_debug.77 = {
	type = country_event
	orphan = yes
	hidden = yes
	title = empty_text
	desc = empty_text
	immediate = {
		if = {
			limit = {
				tag = JAP
				exists = international_organization:kc_kitayama_concord
			}
			kc_refresh_concord_concession_rights = yes
			international_organization:kc_kitayama_concord = {
				remove_variable = kc_overseas_investment_dispatcher_cooldown
				every_international_organization_member = {
					limit = {
						is_ai = yes
						has_reform = government_reform:kc_chartered_buke_house
					}
					remove_variable = kc_house_overseas_construction_cooldown
					remove_variable = kc_house_ship_order_cooldown
					add_gold = 1000
					add_sailors = 0.10
					if = {
						limit = {
							OR = {
								NOT = { has_variable = kc_overseas_concession_licenses_initialized }
								var:kc_overseas_concession_licenses < 1
							}
						}
						set_variable = { name = kc_overseas_concession_licenses value = 1 }
						set_variable = { name = kc_overseas_concession_licenses_initialized value = yes }
						set_variable = { name = kc_overseas_concession_license_available value = yes }
					}
				}
			}
		}
	}
}
