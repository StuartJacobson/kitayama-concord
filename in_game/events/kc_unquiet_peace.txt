namespace = kc_unquiet_peace

# The first transition crisis: peace has removed private war faster than the
# Concord has replaced its local policing, convoy, and coercive institutions.
kc_unquiet_peace.1 = {
	type = country_event
	title = kc_unquiet_peace.1.title
	desc = kc_unquiet_peace.1.desc
	outcome = negative
	image = "gfx/interface/illustrations/disaster/kc_unquiet_peace.dds"

	option = {
		name = kc_unquiet_peace.1.a
		custom_tooltip = {
			text = kc_unquiet_peace_constabulary_choice_tt
			set_variable = {
				name = kc_unquiet_peace_constabulary_preparation
				value = yes
			}
		}
		hidden_effect = {
			international_organization:kc_kitayama_concord ?= {
				every_international_organization_member = {
					limit = {
						has_reform = government_reform:kc_chartered_buke_house
					}
					remove_country_modifier = kc_unquiet_peace_member_disorder
					remove_country_modifier = kc_unquiet_peace_network_mandate
					remove_country_modifier = kc_unquiet_peace_watches_mandate
					add_country_modifier = {
						modifier = kc_unquiet_peace_constabulary_mandate
						years = -1
						mode = replace
					}
				}
			}
		}
		ai_chance = {
			base = 30
			modifier = {
				trigger = {
					international_organization:kc_kitayama_concord ?= {
						international_organization_has_policy = policy:kc_common_arsenals_policy
					}
				}
				add = 60
			}
		
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 0 }
		}
	}

	option = {
		name = kc_unquiet_peace.1.b
		custom_tooltip = {
			text = kc_unquiet_peace_network_choice_tt
			set_variable = {
				name = kc_unquiet_peace_network_preparation
				value = yes
			}
		}
		hidden_effect = {
			international_organization:kc_kitayama_concord ?= {
				every_international_organization_member = {
					limit = {
						has_reform = government_reform:kc_chartered_buke_house
					}
					remove_country_modifier = kc_unquiet_peace_member_disorder
					remove_country_modifier = kc_unquiet_peace_constabulary_mandate
					remove_country_modifier = kc_unquiet_peace_watches_mandate
					add_country_modifier = {
						modifier = kc_unquiet_peace_network_mandate
						years = -1
						mode = replace
					}
				}
			}
		}
		ai_chance = {
			base = 35
			modifier = {
				trigger = {
					international_organization:kc_kitayama_concord ?= {
						OR = {
							international_organization_has_policy = policy:kc_shared_road_fund_policy
							international_organization_has_policy = policy:kc_central_kaido_assessment_policy
						}
					}
				}
				add = 50
			}
		
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 0 }
		}
	}

	option = {
		name = kc_unquiet_peace.1.c
		custom_tooltip = {
			text = kc_unquiet_peace_watches_choice_tt
			set_variable = {
				name = kc_unquiet_peace_watches_preparation
				value = yes
			}
		}
		hidden_effect = {
			international_organization:kc_kitayama_concord ?= {
				every_international_organization_member = {
					limit = {
						has_reform = government_reform:kc_chartered_buke_house
					}
					remove_country_modifier = kc_unquiet_peace_member_disorder
					remove_country_modifier = kc_unquiet_peace_constabulary_mandate
					remove_country_modifier = kc_unquiet_peace_network_mandate
					add_country_modifier = {
						modifier = kc_unquiet_peace_watches_mandate
						years = -1
						mode = replace
					}
				}
			}
		}
		ai_chance = {
			base = 35
			modifier = {
				trigger = {
					international_organization:kc_kitayama_concord ?= {
						OR = {
							international_organization_has_policy = policy:kc_house_retinues_policy
							international_organization_has_policy = policy:kc_registered_retinues_policy
						}
					}
				}
				add = 50
			}
		
			modifier = { kc_ai_japan_follows_house_player = yes  factor = 10 }
		}
	}
}

kc_unquiet_peace.3 = {
	type = country_event
	title = kc_unquiet_peace.3.title
	desc = kc_unquiet_peace.3.desc
	outcome = negative
	image = "gfx/interface/illustrations/disaster/kc_unquiet_peace.dds"

	option = {
		name = kc_unquiet_peace.3.a
		trigger = {
			international_organization:kc_kitayama_concord ?= {
				var:kc_concord_cohesion >= 10
				gold >= {
					value = c:JAP.monthly_income_trade_and_tax
					min = 1
					multiply = 2
				}
			}
		}
		international_organization:kc_kitayama_concord ?= {
			add_gold = {
				value = c:JAP.monthly_income_trade_and_tax
				min = 1
				multiply = -2
			}
			change_variable = {
				name = kc_concord_cohesion
				add = -10
			}
		}
		add_country_modifier = {
			modifier = kc_unquiet_peace_relief_stipends
			years = 5
			mode = replace
		}
		ai_chance = {
			base = 50
		}
